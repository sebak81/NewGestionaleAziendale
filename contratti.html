async function confermaCambioStatoContratto() { const btnSubmit = document.getElementById('btn-conferma-stato-contratto'); const nota = document.getElementById('form-contratto-nota').value; const fileInput = document.getElementById('form-contratto-file'); const con = contrattiCorrenti[contrattoAttivoIndex]; const nuovoStato = statoPendenteContratto; btnSubmit.innerText = "Salvataggio..."; btnSubmit.disabled = true; let nuovoFileUrl = null; if (fileInput.files.length > 0) { btnSubmit.innerText = "Caricamento file..."; nuovoFileUrl = await caricaFileSuStorage(fileInput.files[0], 'CONT_STATO'); } let datiAggiornamento = { stato: nuovoStato }; if (nuovoFileUrl) { datiAggiornamento.file_url = nuovoFileUrl; } if (nota && !con.annotazioni) { datiAggiornamento.annotazioni = nota; } const { error } = await supabaseClient.from('contratti').update(datiAggiornamento).eq('id', con.id); if (!error) { if (nuovoFileUrl) con.file_url = nuovoFileUrl; // --- INIZIO MAGIA FATTURAZIONE --- if (nuovoStato === 'Confermato') { const numeroFattura = con.numero.replace('CONT-', 'FATT-ACC-'); const { data: fattEsistenti } = await supabaseClient.from('fatture').select('id').eq('contratto_id', con.id); if (!fattEsistenti || fattEsistenti.length === 0) { await supabaseClient.from('fatture').insert([{ contratto_id: con.id, commessa_id: con.commessa_id, numero: numeroFattura, stato: 'Da Fare' }]); // Log della creazione automatica await supabaseClient.from('commesse_log').insert([{ commessa_id: con.commessa_id, azione: `Generata in automatico bozza Fattura Acconto: ${numeroFattura}`, autore: UTENTE_CORRENTE }]); } } // --- FINE MAGIA FATTURAZIONE --- await supabaseClient.from('commesse_log').insert([{ commessa_id: con.commessa_id, azione: `Contratto aggiornato in: ${nuovoStato.toUpperCase()}`, nota: nota || null, file_url: nuovoFileUrl, autore: UTENTE_CORRENTE }]); con.stato = nuovoStato; chiudiModaleStatoContratto(); aggiornaDatiSchedaContrattoUI(); caricaCronologia(con.commessa_id); caricaContratti(); } else { alert("Errore aggiornamento: " + error.message); } btnSubmit.innerText = "Conferma e Salva"; btnSubmit.disabled = false; }
